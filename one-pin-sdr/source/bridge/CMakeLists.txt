cmake_minimum_required(VERSION 3.13)
include(${CMAKE_CURRENT_SOURCE_DIR}/pico-ice-sdk/cmake/preinit_pico_ice_sdk.cmake)
set(PICO_SDK_PATH ${CMAKE_CURRENT_SOURCE_DIR}/pico-ice-sdk/lib/pico-sdk)
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(hitl_pico_bridge C CXX ASM)
pico_sdk_init()
add_subdirectory(pico-ice-sdk)
add_executable(hitl_pico_bridge main.c usb_descriptors.c ice_usb_buffered.c)
# Application is C; SDK's unused new/delete object is discarded by section GC.
set_target_properties(hitl_pico_bridge PROPERTIES LINKER_LANGUAGE C)
target_include_directories(hitl_pico_bridge PRIVATE . pico-ice-sdk/lib/tinyuf2/src)
target_link_libraries(hitl_pico_bridge pico_ice_sdk pico_multicore pico_unique_id hardware_watchdog tinyusb_device tinyusb_board)
pico_enable_stdio_usb(hitl_pico_bridge 0)
pico_enable_stdio_uart(hitl_pico_bridge 0)
pico_add_extra_outputs(hitl_pico_bridge)
